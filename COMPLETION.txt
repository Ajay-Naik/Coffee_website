📊 DATABASE STRUCTURE
═════════════════════════════════════════════════════════════════

  orders table:
  • id, full_name, address, city, zip_code
  • payment_method, upi_id, card_number, expiry_date
  • created_at timestamp

  order_items table:
  • id, order_id (links to orders)
  • item_name, item_price, item_quantity
  • created_at timestamp

  contact table:
  • id, name, email, number, message
  • created_at timestamp

  table_booking table:
  • id, name, email, phone_number
  • booking_date, booking_time, guests, status
  • created_at timestamp
